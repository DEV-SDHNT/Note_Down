import React, { useEffect, useRef, useState, useCallback } from "react";

// ----------------------------------------------------------------------------- // useCanvasPan hook (same as before) // -----------------------------------------------------------------------------

export function useCanvasPan( canvasRef, { enabled = true, mode = "transform", // 'transform' | 'callback' friction = 0.92, onPan = null, } = {} ) { const stateRef = useRef({ dragging: false, startX: 0, startY: 0, lastX: 0, lastY: 0, panX: 0, panY: 0, vx: 0, vy: 0, raf: null, });

const applyTransform = useCallback(() => { const el = canvasRef.current; if (!el) return; const { panX, panY } = stateRef.current; el.style.transform = translate(${panX}px, ${panY}px); }, [canvasRef]);

const notify = useCallback(() => { const { panX, panY, lastX, lastY } = stateRef.current; if (typeof onPan === "function") onPan({ x: panX, y: panY, dx: lastX, dy: lastY }); }, [onPan]);

useEffect(() => { const el = canvasRef.current; if (!el || !enabled) return;

const prevPosition = el.style.position;
if (!prevPosition || prevPosition === "static") {
  el.style.position = "relative";
}

el.style.willChange = "transform";
el.style.cursor = "grab";

const state = stateRef.current;

const onPointerDown = (e) => {
  if (e.type === "mousedown" && e.button !== 0) return;
  e.preventDefault();
  const point = e.touches ? e.touches[0] : e;
  state.dragging = true;
  state.startX = point.clientX;
  state.startY = point.clientY;
  state.lastX = 0;
  state.lastY = 0;
  state.vx = 0;
  state.vy = 0;
  el.style.cursor = "grabbing";
};

const onPointerMove = (e) => {
  if (!state.dragging) return;
  const point = e.touches ? e.touches[0] : e;
  const dx = point.clientX - state.startX;
  const dy = point.clientY - state.startY;
  const ddx = dx - state.lastX;
  const ddy = dy - state.lastY;
  state.lastX = dx;
  state.lastY = dy;
  state.panX += ddx;
  state.panY += ddy;
  state.vx = ddx;
  state.vy = ddy;
  if (mode === "transform") applyTransform();
  notify();
};

const onPointerUp = () => {
  if (!state.dragging) return;
  state.dragging = false;
  el.style.cursor = "grab";
  if (friction < 0.999 && (Math.abs(state.vx) > 0.5 || Math.abs(state.vy) > 0.5)) {
    const step = () => {
      state.vx *= friction;
      state.vy *= friction;
      state.panX += state.vx;
      state.panY += state.vy;
      if (mode === "transform") applyTransform();
      notify();
      if (Math.abs(state.vx) > 0.5 || Math.abs(state.vy) > 0.5) {
        state.raf = requestAnimationFrame(step);
      } else {
        state.raf = null;
      }
    };
    if (!state.raf) state.raf = requestAnimationFrame(step);
  }
};

const onWheel = (e) => {
  e.preventDefault();
  const deltaX = e.shiftKey ? e.deltaY : e.deltaX;
  const deltaY = e.shiftKey ? 0 : e.deltaY;
  state.panX -= deltaX;
  state.panY -= deltaY;
  state.vx = -deltaX;
  state.vy = -deltaY;
  if (mode === "transform") applyTransform();
  notify();
};

el.addEventListener("mousedown", onPointerDown);
window.addEventListener("mousemove", onPointerMove, { passive: false });
window.addEventListener("mouseup", onPointerUp);
el.addEventListener("touchstart", onPointerDown, { passive: false });
window.addEventListener("touchmove", onPointerMove, { passive: false });
window.addEventListener("touchend", onPointerUp);
el.addEventListener("wheel", onWheel, { passive: false });

return () => {
  if (state.raf) cancelAnimationFrame(state.raf);
  el.removeEventListener("mousedown", onPointerDown);
  window.removeEventListener("mousemove", onPointerMove);
  window.removeEventListener("mouseup", onPointerUp);
  el.removeEventListener("touchstart", onPointerDown);
  window.removeEventListener("touchmove", onPointerMove);
  window.removeEventListener("touchend", onPointerUp);
  el.removeEventListener("wheel", onWheel);
  el.style.cursor = "";
  el.style.willChange = "";
};

}, [canvasRef, enabled, mode, friction, applyTransform, notify]);

const getPan = () => { const { panX, panY } = stateRef.current; return { x: panX, y: panY }; };

const setPan = (x, y) => { stateRef.current.panX = x; stateRef.current.panY = y; if (mode === "transform") applyTransform(); notify(); };

return { getPan, setPan }; }

// ----------------------------------------------------------------------------- // InfiniteCanvas — draws a repeating grid, supports infinite panning. // -----------------------------------------------------------------------------

export default function InfiniteCanvas() {
const canvasRef = useRef(null);
const { getPan } = useCanvasPan(canvasRef, { mode: "callback", friction: 0.92, onPan: draw });


function draw({ x, y }) {
const canvas = canvasRef.current;
if (!canvas) return;
const ctx = canvas.getContext("2d");
const DPR = window.devicePixelRatio || 1;
const width = canvas.clientWidth;
const height = canvas.clientHeight;
canvas.width = width * DPR;
canvas.height = height * DPR;
ctx.scale(DPR, DPR);


ctx.fillStyle = "#0f172a";

ctx.fillRect(0, 0, width, height);

const gridSize = 40;
const offsetX = x % gridSize;
const offsetY = y % gridSize;

ctx.strokeStyle = "rgba(255,255,255,0.07)";
for (let gx = -gridSize; gx < width + gridSize; gx += gridSize) {
  ctx.beginPath();
  ctx.moveTo(gx + offsetX, 0);
  ctx.lineTo(gx + offsetX, height);
  ctx.stroke();
}
for (let gy = -gridSize; gy < height + gridSize; gy += gridSize) {
  ctx.beginPath();
  ctx.moveTo(0, gy + offsetY);
  ctx.lineTo(width, gy + offsetY);
  ctx.stroke();
}

ctx.fillStyle = "rgba(99,102,241,0.9)";
ctx.beginPath();
ctx.arc(width / 2, height / 2, 8, 0, Math.PI * 2);
ctx.fill();

ctx.fillStyle = "rgba(255,255,255,0.8)";
ctx.font = "13px system-ui, sans-serif";
ctx.fillText(`Pan anywhere — Infinite grid. Pan: ${Math.round(x)}, ${Math.round(y)}`, 12, 20);

}

useEffect(() => {
const canvas = canvasRef.current;
if (canvas) draw({ x: 0, y: 0 });
}, []);

return (
<div className="p-4 max-w-4xl mx-auto">
<div className="rounded-lg shadow bg-white/5 overflow-hidden">
<canvas ref={canvasRef} style={{ touchAction: "none", width: "100%", height: "500px", display: "block" }} />
</div> </div> ); }
